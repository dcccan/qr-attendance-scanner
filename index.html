/** * Executes on Form Submit. * Evaluates attendance: * - If "Yes, I'll be there": Generates UUID token, emails QR code, marks "YES" in Col J. * - If "Sorry, can't make it": Marks "N/A (Not Attending)" in Col I & J. */ function onFormSubmit(e) { const sheet = e.range.getSheet(); const row = e.range.getRow(); // 1-based column indices for writing results back const TOKEN_COL = 9; // Col I: QR Token const EMAIL_SENT_COL = 10; // Col J: Email Sent // Safely extract values directly from the event object 'e' // Fallbacks provided using array values if e.namedValues isn't available const canAttend = e.namedValues ? (e.namedValues['Can you attend?'] ? e.namedValues['Can you attend?'][0] : '') : (e.values[2] || ''); const email = e.namedValues ? (e.namedValues['Email address'] ? e.namedValues['Email address'][0] : '') : (e.values[1] || ''); const name = e.namedValues ? (e.namedValues['Full Name'] ? e.namedValues['Full Name'][0] : '') : (e.values[3] || ''); const cleanAttend = String(canAttend).trim(); // Case 1: Participant cannot attend if (cleanAttend.toLowerCase().includes("can't make it") || cleanAttend.toLowerCase().includes("sorry")) { sheet.getRange(row, TOKEN_COL).setValue("N/A (Not Attending)"); sheet.getRange(row, EMAIL_SENT_COL).setValue("N/A (Not Attending)"); return; } // Case 2: Participant confirmed attendance ("Yes, I'll be there") if (cleanAttend.toLowerCase().startsWith("yes")) { const qrToken = Utilities.getUuid(); sheet.getRange(row, TOKEN_COL).setValue(qrToken); if (email.trim()) { const qrUrl = "https://api.qrserver.com/v1/create-qr-code/?size=250x250&data=" + encodeURIComponent(qrToken); const qrBlob = UrlFetchApp.fetch(qrUrl).getBlob().setName("QRCode.png"); MailApp.sendEmail({ to: email.trim(), subject: "Your Event Check-In QR Code", htmlBody: `

Hi ${name},

` + `

Thank you for confirming your attendance! Please present the attached QR code at check-in.

` + `

Token ID: ${qrToken}

`, attachments: [qrBlob] }); sheet.getRange(row, EMAIL_SENT_COL).setValue('YES'); } } } /** * Web App Endpoint for processing check-ins from the HTML scanner page. */ function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(10000); let res; try { let scannedToken = ''; let scannerUser = ''; // Handle incoming parameters safely across all mobile browsers if (e.parameter && e.parameter.id) { scannedToken = e.parameter.id.trim(); scannerUser = (e.parameter.scannedBy || 'Gate 1').trim(); } else if (e.postData && e.postData.contents) { try { const payload = JSON.parse(e.postData.contents); scannedToken = (payload.id || '').trim(); scannerUser = (payload.scannedBy || 'Gate 1').trim(); } catch (jsonErr) { // Fallback if URLSearchParams posted raw string scannedToken = e.postData.contents.trim(); } } // Guard Clause: Prevent blank/empty string matching against empty sheet cells if (!scannedToken || scannedToken === "N/A (Not Attending)") { res = { status: 'unknown' }; return ContentService.createTextOutput(JSON.stringify(res)) .setMimeType(ContentService.MimeType.JSON); } const ss = SpreadsheetApp.getActive(); const respSheet = ss.getSheetByName('Form responses 1'); const respData = respSheet.getDataRange().getValues(); // 0-based array indices for 'Form responses 1' const TOKEN_INDEX = 8; // Col I (QR Token) const NAME_INDEX = 3; // Col D (Full Name) const AFFIL_INDEX = 4; // Col E (Affiliation) const ROLE_INDEX = 5; // Col F (Role/Position) // Search for registrant matching scanned token const person = respData.find(row => String(row[TOKEN_INDEX]).trim() === scannedToken); const attSheet = ss.getSheetByName('Attendance'); const attData = attSheet.getDataRange().getValues(); if (!person) { res = { status: 'unknown' }; } else if (attData.some(row => String(row[0]).trim() === scannedToken)) { res = { status: 'duplicate', name: person[NAME_INDEX] }; } else { attSheet.appendRow([ scannedToken, person[NAME_INDEX], person[AFFIL_INDEX], person[ROLE_INDEX], new Date(), scannerUser ]); res = { status: 'ok', name: person[NAME_INDEX], affiliation: person[AFFIL_INDEX] }; } } catch (err) { res = { status: 'error', message: err.toString() }; } finally { lock.releaseLock(); } return ContentService.createTextOutput(JSON.stringify(res)) .setMimeType(ContentService.MimeType.JSON); }
